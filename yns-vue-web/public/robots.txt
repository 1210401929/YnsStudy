//校验是否有填报资格
function checkIsFillQualiFication() {
    //update Y 2025-4-11 主模板设置增加模板类型,财政版和单位版  如果模板是单位版,则不校验公开单位和公开时间的资格限制
    let mainCat = getMainFation($DS.getCtrl("STEP_步骤").info);
    //mainCat["MODELTYPE"] 是 模板类型  1:财政版,0:单位版
    if(mainCat["MODELTYPE"] == "0"){
        //是否是单位版模板
        modelIsAgencyType  = true;
        return true;
    }
    //根据条件查询该用户单位是否不公开,如果不公开,不允许填报
    let isReturnRes = isReturn();
    if (!isReturnRes.noError) {
        isInsert = false;
        //当设置不公开时,作废之前生成的数据   将ISVOID改为'1'
        $DS.exeSql(`update GFA_ANGENCY_STATUS
                    set ISVOID = '1'
                    where year = '${year}'
                      and admdiv = '${admdiv}'
                      and AGENCYID = '${getUpdividInfo().updivid}'
                      and GFATYPE = '${GFATYPE}'`);
        if (isReturnRes.errMsg) {
            alert(isReturnRes.errMsg, "pageCloseY");
        }

        return false;
    }
    let dateInfo = getLastDateAndOverDate();
    if (!dateInfo) {
        isInsert = false;
        alert("未设置公开开始时间,不允许填报!", "pageCloseY");
        return false;
    }
    let newDateTime = $DS.util.timeFormate(new Date(), "yyyy-MM-dd HH:mm:ss");
    if (getDaysBetween(newDateTime, dateInfo["RESPOND_DATE"]) > 0) {
        isInsert = false;
        alert("时间未开始,不允许填报!", "pageCloseY");
        return false;
    }
    return true;
}

//根据条件查询该用户单位是否不公开,如果不公开,不允许填报
function isReturn() {
    let sql = new String();
    let result;
    let res = {noError: true, errMsg: ""};
    if (userType == -1) {
        res.noError = false;
        res.errMsg = "管理员不能进行填报，请使用具体单位账号!";
    }

    //政府公开为财政用户操作,  需要查区划设置
    if (["GFA11", "GFA12"].indexOf(GFATYPE) != -1)
        sql = `select *
               from GFA_REPORT_ADMDIV
               where year = '${year}' and MOF_DIV_ID = '${admdiv}' and GFA_TYPE_CODE='${GFATYPE}'`;
    //查单位设置
    else if (["GFA01", "GFA02"].indexOf(GFATYPE) != -1){
        //先查区划设置,再查单位设置
        sql = `select *
               from GFA_REPORT_ADMDIV
               where year = '${year}' and MOF_DIV_ID = '${admdiv}' and GFA_TYPE_CODE='${GFATYPE}'`;
        result = $DS.selectBySql(VUECFG.appId, sql).result;
        if (!result || !result[0]) {
            res.noError = false;
            res.errMsg = "该区划未设置是否公开,不允许填报!";
            return res;
        }
        //如果该单位、区划不公开 退出   默认为公开
        if (result && result[0] && result[0]["IS_PUBLIC"] && result[0]["IS_PUBLIC"] == "0") {
            res.noError = false;
            //2024-10-17 update Y  当该单位不公开时,将弹出页面用于输入不公开原因
            res.errMsg = null;
            $DS.showPage(`freeFromView.html?PAGEID=F068D1A30F4A4F0AB54CEAC91E19F9A8&PAGETITLE=【G1-1-O】不公开单位填写原因页面&APPID=GFA&year=${year}&agencyId=${agencyId}&GFATYPE=${GFATYPE}`, `填写不公开原因`, `60%`, `75%`);
            return res;
        }
        sql = `select *
               from GFA_REPORT_AGENCY
               where year = '${year}' and agency_id = '${agencyId}' and GFA_TYPE_CODE='${GFATYPE}'`;
    }

    result = $DS.selectBySql(VUECFG.appId, sql).result;
    if (!result || !result[0]) {
        res.noError = false;
        res.errMsg = "该区划单位未设置是否公开,不允许填报!";
        return res;
    }
    //如果该单位、区划不公开 退出   默认为公开
    if (result && result[0] && result[0]["IS_PUBLIC"] && result[0]["IS_PUBLIC"] == "0") {
        res.noError = false;
        //2024-10-17 update Y  当该单位不公开时,将弹出页面用于输入不公开原因
        res.errMsg = null;
        $DS.showPage(`freeFromView.html?PAGEID=F068D1A30F4A4F0AB54CEAC91E19F9A8&PAGETITLE=【G1-1-O】不公开单位填写原因页面&APPID=GFA&year=${year}&agencyId=${agencyId}&GFATYPE=${GFATYPE}`, `填写不公开原因`, `60%`, `75%`);
        return res;
    }
    //给记载单位信息设置表的数据赋值
    agencyMainData = result[0];
    return res;
}

//listView点击事件
function listView_click(obj, val) {
    debugger;
    //点击前校验
    if(!clickCheckBf()) return false;
    //构造参数
    putTopWinPms();
    //设置css类名
    setListCssFun();

    $DS.putPms("listPms", val);
    //记录节点信息 update Y 2025-2-20
    recordNodeInformation();
    let stepInfo = $DS.getCtrl("STEP_步骤").info;
    //判断是否是最后一步,如果是,执行推送参数方法
    if (stepInfo.ds_step_active == stepInfo.ds_step_cfg.length - 1)
        getPdfViewPageCfg();
    iframe_complete();

    //设置css类名
    function setListCssFun(){
        //----------------设置当前选中css类名--------------------
        $(".listSelected").removeClass("listSelected")
        if (val.$_event)
            $(val.$_event.currentTarget).find("div:first").addClass("listSelected");
    }

    //点击前校验
    function clickCheckBf(){
        //类似于一个后门,当模板内保存自动调用点击下一个步骤时,则跳过校验
        if($WINTOP["off_listview_click_check"]){
            $WINTOP["off_listview_click_check"] = false;
            return true;
        }
        if(pageMainCat && pageMainCat.ISFORCECHECK && pageMainCat.ISFORCECHECK == "1"){
        //if(1){
            if(["06", "5", "04", "03", "3"].indexOf(taskData["SUBSTATUS"]) != -1){
                let currentIndex = val.oneDataIndex;
                let lastIndex = currentIndex -1;
                if(obj.ds_listview[lastIndex] && obj.ds_listview[lastIndex].class_!="listSuccess"){
                    alert("请按顺序逐项完善信息并保存!");
                    return false;
                }
            }
        }
        return true;
    }
}

//记录节点信息  打开填报信息时,定位到最后点击的节点
function recordNodeInformation() {
    //步骤条节点
    let stepPms = $DS.getPms("stepPms");
    //listView节点
    let listPms = $DS.getPms("listPms");
    if (stepPms && !stepPms["GUID"])
        stepPms = JSON.parse(stepPms);
    //目前只记录填报状态的节点缓存  文本,附件,可下载附件  去掉该条件,则记录所有节点无任何限制
    if (["T"].indexOf(stepPms["REPORTTYPE"]) != -1 || stepPms.isSub) {
        //存储到浏览器缓存
        localStorage.setItem(getCacheKeys().stepCacheKey, stepPms["GUID"]);
        localStorage.setItem(getCacheKeys().listViewCacheKey, listPms["GUID"]);
    }
}

//存储缓存key获取入口
function getCacheKeys(){
    let stepCacheKey = "cache_stepPmsGuid_"+taskData.GUID;
    let listViewCacheKey = "cache_listPmsGuid_"+taskData.GUID;
    return {
        stepCacheKey,listViewCacheKey
    }
}

//根据缓存信息定位到某节点
function locationByCaChe(type) {
    if (type == "step") {
        let stepPmsGuid = localStorage.getItem(getCacheKeys().stepCacheKey);
        if (stepPmsGuid) {
            let stepInfo = $DS.getCtrl("STEP_步骤").info;
            //缓存信息与步骤条配置信息比对  如果相同,则定位到该步骤
            stepInfo.ds_step_cfg.forEach((cfg, index) => {
                let data = JSON.parse(cfg.url);
                if (data["GUID"] == stepPmsGuid) {
                    stepInfo.ds_step_active = index;
                }
            });
            return true;
        } else {
            return false;
        }
    } else if (type == "listView") {
        let listPmsGuid = localStorage.getItem(getCacheKeys().listViewCacheKey);
        if (listPmsGuid) {
            let iden = false;
            let listViewInfo = $DS.getCtrl("LISTVIEW_操作明细步骤").info;
            listViewInfo.ds_listview.forEach((item, index) => {
                if (item["GUID"] == listPmsGuid) {
                    iden = true;
                    //推送参数
                    $DS.putPms("listPms", listViewInfo.ds_listview[index]);
                    //模拟点击
                    var listViewIntertVal = setInterval(function () {
                        if (VUECFG.$refs[listViewInfo.ds_id]) {
                            VUECFG.$refs[listViewInfo.ds_id].$nextTick(function () {
                                //模拟点击
                                $(`.font-enl:eq(${index})`).click();
                                //将元素移动到可视区域
                                $(`.font-enl:eq(${index})`).get(0).scrollIntoView({
                                    behavior: "smooth",
                                    block: "center"
                                });
                            });
                            window.clearInterval(listViewIntertVal);
                        }
                    });
                }
            });
            return iden;
        } else {
            return false;
        }
    }
    return false;
}

        let hideArr_ = new Array();

                             from GFA_REPORT_TEMPLET
                             where GFACAT = '${gfaCat}'
                               and HIDECONDITION is not null`;

            result.result.forEach(item => {
                if (item["HIDECONDITION"] && eval($DS.util.replace(item["HIDECONDITION"]))) {
                    hideArr_.push(item["GUID"]);
                }
            });
        } catch (e) {
            //console.error(e);
            hideArr_ = [];
            return hideArr_;
        }

        return hideArr_;
    }
    //查询是否隐藏报表  update Y 2025-3-20
    function isHideReport(reportLimit){
        //如果开启了不拼接报表  并且不拼接条件没有值
        if(reportLimit["noMontageR"] && reportLimit["noMontageR"]=="true" && (!reportLimit["noMontageRCondition"] || !reportLimit["noMontageRCondition"].trim())){
            return true;
            //如果不拼接条件有值
        }else if(reportLimit["noMontageRCondition"] && reportLimit["noMontageRCondition"].trim()){
            let base = new Base64();
            let noMontageRCondition = base.decode(reportLimit["noMontageRCondition"]);
            if(eval($DS.util.replace(noMontageRCondition))){
                return true;
            }
        }
        return false;
    }

//加工listView数据
//isCheck参数:当时true时,表示需要用这个方法判断当前大节点是否全部完成
function machinData(data,isCheck) {

    let currentIndex = 0;

    let completeDetail = taskData.COMPLETE_DETAIL;
    completeDetail = completeDetail ? JSON.parse(completeDetail) : {};
    let newData = data.map((item,index)=>{
        let g = item["GUID"];
        if(completeDetail[g] && completeDetail[g].isSave){
            item.class_ = "listSuccess";
            currentIndex++;
        }

        item.oneDataIndex = index;
        return item;
    });

    //---------------修改下方LABEL进度文字--------------------
    let stepPms = $DS.getPms("stepPms");
    if (stepPms && !stepPms["GUID"])
        stepPms = JSON.parse(stepPms);
    let labelInfo = $DS.getCtrl("LABEL_进度").info;
    let allLength = newData.length;
    //如果是模板填报  或者是子模版
    if(stepPms.REPORTTYPE==="T" || stepPms.isSub){
        labelInfo.ds_show = true;

        let labelHtml = `<div style="display: flex; justify-content: center; align-items: center; width: 100%; height: 100%; box-sizing: border-box; color: rgba(85, 115, 135, 1); font-size: 1rem;">
                        <div style="width: 80%; height: 100%; display: flex; justify-content: space-between; align-items: center;">
                            <div class="left-text">
                            已完成
                            </div>
                            <div class="right-text">
                                ${currentIndex} / ${allLength} 项
                            </div>
                        </div>
                    </div>`;
        labelInfo.ds_expression = labelHtml;
        //赋值给组件,步骤条变更时,控制逻辑会用到
        labelInfo.currentIndex = currentIndex;
        labelInfo.allLength = allLength;
    }else{
        labelInfo.ds_show = false;
    }
    if(isCheck){
        //判断是否开启了校验
        if(pageMainCat && pageMainCat.ISFORCECHECK && pageMainCat.ISFORCECHECK == "1"){
            //判断是否是需要校验的状态
            if(["06", "5", "04", "03", "3"].indexOf(taskData["SUBSTATUS"]) != -1 && (stepPms.REPORTTYPE==="T" || stepPms.isSub)){
                //判断是否完成了所有步骤
                if(currentIndex === allLength){
                    return false;
                    //返回true,表示触发校验拦截,不允许下一步操作
                }else
                    return true;
            }else{
                return false;
            }
        }else{
            return false
        }

    }else{
        return newData;
    }

}

//获取批复日期和逾期时间
function getLastDateAndOverDate() {
    let isAgency = $DS.getPms("URL_AGENCY");
    let sql = new String();
    let result;
    let dateField = ['RESPOND_DATE', 'RESPOND_LASTDATE', 'OVERT_DATE'];
    let selectStr = dateField.map(field => ` to_char(${field},'yyyy-MM-dd HH24:mi:ss') as ${field} `).join(',');

    let updivIdInfo = getUpdividInfo();
    //单位用户
    if (isAgency == "true") {
        let agencyId = updivIdInfo.updivid;
        sql = `select ${selectStr}
               from GFA_APPROVAL_AGENCY
               where APPID = '${GFATYPE}' and mof_div_code = '${admdivCode}' and AGENCY_ID = '${agencyId}' and year =
                     '${year}'`;
        result = $DS.selectBySql(VUECFG.appId, sql);
        //如果本单位已设置如期
        if (result && !result.isError && result.result && result.result[0] && result.result.length > 0 && result.result[0]) {
            return {
                RESPOND_DATE: strTimeForMT(result.result[0]["RESPOND_DATE"] || ""),
                RESPOND_LASTDATE: strTimeForMT(result.result[0]["RESPOND_LASTDATE"] || ""),
                OVERT_DATE: strTimeForMT(result.result[0]["OVERT_DATE"] || ""),
            }
            //找不到,向上级部门找
        } else {
            result = $DS.selectBySql(VUECFG.appId, `select superguid
                                                    from SSO_V_PUBAGENCY
                                                    where guid = '${agencyId}' and year = '${year}'`);
            if (!result || result.isError || !result.result || result.result.length == 0)
                return false;
            let updivid = result.result[0]["SUPERGUID"];
            result = $DS.selectBySql(VUECFG.appId, `select ${selectStr}
                                                    from GFA_APPROVAL_AGENCY
                                                    where APPID = '${GFATYPE}' and year = '${year}'
                                                      and mof_div_code = '${admdivCode}'
                                                      and AGENCY_ID = '${updivid}'`);

            if (result && !result.isError && result.result && result.result[0] && result.result.length > 0 && result.result[0]) {
                return {
                    RESPOND_DATE: strTimeForMT(result.result[0]["RESPOND_DATE"] || ""),
                    RESPOND_LASTDATE: strTimeForMT(result.result[0]["RESPOND_LASTDATE"] || ""),
                    OVERT_DATE: strTimeForMT(result.result[0]["OVERT_DATE"] || ""),
                }
            } else {
                return false;
            }
        }
        //部门用户
    } else {
        let updivId = updivIdInfo.updivid;
        result = $DS.selectBySql(VUECFG.appId, `select ${selectStr}
                                                from GFA_APPROVAL_DEPART
                                                where APPID = '${GFATYPE}' and year = '${year}'
                                                  and mof_div_code = '${admdivCode}'
                                                  and agency_id = '${updivId}'`);
        if (result && !result.isError && result.result && result.result[0] && result.result.length > 0 && result.result[0]) {
            return {
                RESPOND_DATE: strTimeForMT(result.result[0]["RESPOND_DATE"] || ""),
                RESPOND_LASTDATE: strTimeForMT(result.result[0]["RESPOND_LASTDATE"] || ""),
                OVERT_DATE: strTimeForMT(result.result[0]["OVERT_DATE"] || ""),
            }
        } else {
            return false;
        }
    }
}

/**
 *格式化数据库字段
 */
function strTimeForMT(str) {
    if (!str) return "";
    const finalTime = str.replace(/-/g, '/').replace('T', ' ').replace('.000+0000', '')
    return finalTime.toString();
}

/**
 * 倒计时
 * `${days}天${hours}小时${minutes}分钟${seconds}秒`
 * @param endDateStr
 * @param fun
 */
function timeUp(endDateStr, fun, pms) {
    //结束时间
    var endDate = new Date(endDateStr);
    //当前时间
    var nowDate = new Date();
    //相差的总秒数
    var totalSeconds = parseInt((nowDate - endDate) / 1000);
    //天数
    var days = Math.floor(totalSeconds / (60 * 60 * 24));
    //取模（余数）
    var model = totalSeconds % (60 * 60 * 24);
    //小时数
    var hours = Math.floor(model / (60 * 60));
    model = model % (60 * 60);
    //分钟
    var minutes = Math.floor(model / 60);
    //秒
    var seconds = model % 60;
    //输出到页面
    fun({days, hours, minutes, seconds}, pms)
    //延迟一秒执行自己
    setTimeout(function () {
        timeUp(endDateStr, fun, pms);
    }, 1000)
}

//获取系统当前时间
function getCurrentDateTime() {
    const now = new Date();
    const year = now.getFullYear();
    const month = String(now.getMonth() + 1).padStart(2, '0'); // 月份从0开始，需要加1
    const day = String(now.getDate()).padStart(2, '0');
    const hours = String(now.getHours()).padStart(2, '0');
    const minutes = String(now.getMinutes()).padStart(2, '0');
    const seconds = String(now.getSeconds()).padStart(2, '0');
    return `${year}年${month}月${day}日 ${hours}时${minutes}分${seconds}秒`;
    // return `${year}/${month}/${day} ${hours}:${minutes}:${seconds}`;
}

/**
 * 计算两个日期之间的天数
 *  date1  开始日期 yyyy-MM-dd
 *  date2  结束日期 yyyy-MM-dd
 *  如果日期相同 返回一天 开始日期大于结束日期，返回0
 */
function getDaysBetween(date1, date2) {
    var startDate = Date.parse(date1);
    var endDate = Date.parse(date2);
    if (startDate > endDate) {
        return 0;
    }
    if (startDate == endDate) {
        return 1;
    }
    var days = (endDate - startDate) / (1 * 24 * 60 * 60 * 1000);
    return days;
}

    if (result && result.result && result.result[0] && result.result[0]["ISCONSTRAINT"] && result.result[0]["ISCONSTRAINT"] == "1") {
        let sxCfg = {isSx: "1", cfg: {}};
        if (result.result[0]["CONSTRAINTINFO"]) {
            sxCfg.cfg = JSON.parse(result.result[0]["CONSTRAINTINFO"]);
        }
        let key = `${gfaType}_sxCfg`;
        $WINTOP[key] = sxCfg;
        $WINTOP["sxCfg"] = sxCfg;
        return sxCfg;
    } else {
        return {};
    }
}

//根据类型弹出抽屉组件 可以直接传入url  update Y 2025-2-27增加功能
function showDrawerByType(url, type) {
    let type2url = {
        showSubData: "freeFromView.html?PAGEID=522D5EE6402F43DEA79CF63962F7B710&PAGETITLE=【G1-1-2-1】文本查看下级填报界面&APPID=GFA"
    }
    type = type || "Y";
    url = url || type2url[type] || "";

    if (url) {
        $drawer.show('DRAWER_抽屉');
        let stopIden = setInterval(() => {
            let iframeInfo = $DS.getCtrl("IFRAME_网页").info;
            if (url && iframeInfo) {
                clearInterval(stopIden);
                $(`#${iframeInfo.ds_id}_iframe`).attr("src", "");
                $(`#${iframeInfo.ds_id}_iframe`).attr("src", url);
            }
        })
    }
}
